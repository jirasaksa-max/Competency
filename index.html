// Data query schema matching frontend bindings SELECT c.competency_name, COUNT(CASE WHEN eval.score < eval.expected_level THEN 1 END) AS gap_count, COUNT(eval.id) AS total_evaluated, ROUND((COUNT(CASE WHEN eval.score < eval.expected_level THEN 1 END) * 100.0 / COUNT(eval.id)), 2) AS gap_percentage FROM competency_evaluations eval JOIN competencies c ON eval.competency_id = c.id GROUP BY c.id, c.competency_name;
